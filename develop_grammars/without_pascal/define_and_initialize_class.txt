### LEXER

// DELETE THIS CONTENT IF YOU PUT COMBINED GRAMMAR IN Parser TAB
lexer grammar ExprLexer;

AND : 'and' ;
OR : 'or' ;
NOT : 'not' ;
EQ : '=' ;
COMMA : ',' ;
SEMI : ';' ;
LPAREN : '(' ;
RPAREN : ')' ;
LCURLY : '{' ;
RCURLY : '}' ;

INT : [0-9]+ ;
ID: [a-zA-Z_][a-zA-Z_0-9]* ;
WS: [ \t\n\r\f]+ -> skip ;

CLASS : 'class' ;
COLON : ':' ;
DOT : '.' ;

---

### PARSER

parser grammar ExprParser;
options { tokenVocab=ExprLexer; }

program
    : (classDefinition | def | stat)* EOF
    ;

stat: ID '=' expr
    | expr
    ;

classDefinition
    : CLASS ID '(' ')' (stat)*
    ;

callClass
    : ID '(' ')'
    | CLASS ID '(' ')'
    | ID COLON '=' ID
    | ID COLON '=' ID DOT ID
    ;

def : ID '(' ID (',' ID)* ')' '{' stat* '}' ;

expr: ID
    | INT
    | func
    | callClass
    | 'not' expr
    | expr 'and' expr
    | expr 'or' expr
    ;

func : ID '(' expr (',' expr)* ')' ;
